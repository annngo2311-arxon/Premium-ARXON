-- ===================== SAFE PARENT HELPER =====================
local function getSafeGuiParent()
local playerGui = nil
pcall(function()
    playerGui = LocalPlayer:WaitForChild("PlayerGui", 5)
end)
if playerGui then
    return playerGui
end
local success, parent = pcall(function()
    if type(gethui) == "function" then
        return gethui()
    end
    return game:GetService("CoreGui")
end)
if success and parent then
    return parent
end
return nil
end

-- Dọn dẹp GUI cũ nếu đã tồn tại
local targetParent = getSafeGuiParent()
if not targetParent then
    return
end
-- Remove stale copies from both common GUI containers to prevent stacking.
for _, container in ipairs({LocalPlayer:FindFirstChildOfClass("PlayerGui"), targetParent}) do
    if container then
        pcall(function()
            for _, child in ipairs(container:GetChildren()) do
                if child.Name == "AXN" then child:Destroy() end
            end
        end)
    end
end

-- ===================== DRAG HELPER =====================
local function makeDraggable(frame, handle)
local dragging, dragStart, startPos
handle.InputBegan:Connect(function(input)
if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
dragging = true
dragStart = input.Position
startPos = frame.Position

local connection  
        connection = UserInputService.InputChanged:Connect(function(moveInput)  
            if dragging and (moveInput.UserInputType == Enum.UserInputType.MouseMovement or moveInput.UserInputType == Enum.UserInputType.Touch) then  
                local delta = moveInput.Position - dragStart  
                frame.Position = UDim2.new(  
                    startPos.X.Scale, startPos.X.Offset + delta.X,  
                    startPos.Y.Scale, startPos.Y.Offset + delta.Y  
                )  
            end  
        end)  
          
        input.Changed:Connect(function()  
            if input.UserInputState == Enum.UserInputState.End then  
                dragging = false  
                if connection then connection:Disconnect() end  
            end  
        end)  
    end  
end)

end

-- ===================== FIX LAG =====================
local fixLagActive = false
local function fixLag()
local removeClasses = {
"ParticleEmitter", "Trail", "Smoke", "Fire", "Sparkles",
"Beam", "Explosion", "Highlight", "Decal", "Texture",
"PointLight", "SpotLight", "SurfaceLight", "BlurEffect",
"ColorCorrectionEffect", "SunRaysEffect", "DepthOfFieldEffect",
"BloomEffect", "Atmosphere", "Sky", "Clouds"
}

for _, obj in ipairs(workspace:GetDescendants()) do  
    local destroyed = false  
    for _, className in ipairs(removeClasses) do  
        if obj:IsA(className) then  
            obj:Destroy()  
            destroyed = true  
            break  
        end  
    end  
    if not destroyed and obj:IsA("BasePart") then  
        obj.CastShadow = false  
    end  
end  

for _, obj in ipairs(Lighting:GetChildren()) do  
    obj:Destroy()  
end  

Lighting.GlobalShadows = false  
Lighting.FogEnd = 999999  
Lighting.Brightness = 2  

if not fixLagActive then  
    fixLagActive = true  
    local conn = workspace.DescendantAdded:Connect(function(obj)  
        if not fixLagActive then return end  
        for _, className in ipairs(removeClasses) do  
            if obj:IsA(className) then  
                task.defer(function()  
                    if obj and obj.Parent then obj:Destroy() end  
                end)  
                break  
            end  
        end  
        if obj:IsA("BasePart") then  
            obj.CastShadow = false  
        end  
    end)  
    table.insert(activeConnections, conn)  
end

end
FixLagBtn.MouseButton1Click:Connect(fixLag)


-- ===================== KHÔI PHỤC ĐỒ HỌA =====================
local fastModeSnapshot = {parts = {}, effects = {}, globalShadows = nil}
local function rememberFastModeGraphics()
    fastModeSnapshot = {parts = {}, effects = {}, globalShadows = Lighting.GlobalShadows}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            fastModeSnapshot.parts[obj] = obj.CastShadow
        elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") then
            pcall(function() fastModeSnapshot.effects[obj] = obj.Enabled end)
        end
    end
end
local restoreFastModeGraphics = function()
    for obj, value in pairs(fastModeSnapshot.parts) do
        pcall(function() if obj and obj.Parent then obj.CastShadow = value end end)
    end
    for obj, value in pairs(fastModeSnapshot.effects) do
        pcall(function() if obj and obj.Parent then obj.Enabled = value end end)
    end
    if fastModeSnapshot.globalShadows ~= nil then
        pcall(function() Lighting.GlobalShadows = fastModeSnapshot.globalShadows end)
    end
end
RestoreGraphicsBtn.MouseButton1Click:Connect(restoreFastModeGraphics)

-- ===================== REMOVE TREES =====================
local removeTreesActive = false
local function isTreeTrunkOnly(obj)
if not obj:IsA("BasePart") then return false end
local name = string.lower(obj.Name)
if string.find(name, "leaf") or string.find(name, "leaves") or string.find(name, "bush") or string.find(name, "plant") then
return false
end
if string.find(name, "tree") or string.find(name, "trunk") or string.find(name, "branch") then
return true
end
if obj.Parent then
local parentName = string.lower(obj.Parent.Name)
if string.find(parentName, "leaf") or string.find(parentName, "leaves") or string.find(parentName, "bush") or string.find(parentName, "plant") then
return false
end
if string.find(parentName, "tree") then
return true
end
end
return false
end

local function removeTrees()
for _, obj in ipairs(workspace:GetDescendants()) do
if obj ~= LocalPlayer.Character and not obj:IsDescendantOf(LocalPlayer.Character) then
if isTreeTrunkOnly(obj) then obj:Destroy() end
end
end

if not removeTreesActive then  
    removeTreesActive = true  
    local conn = workspace.DescendantAdded:Connect(function(obj)  
        if not removeTreesActive then return end  
        if obj == LocalPlayer.Character or obj:IsDescendantOf(LocalPlayer.Character) then return end  
        if isTreeTrunkOnly(obj) then  
            task.defer(function()  
                if obj and obj.Parent then obj:Destroy() end  
            end)  
        end  
    end)  
    table.insert(activeConnections, conn)  
end

end
RemoveTreesBtn.MouseButton1Click:Connect(removeTrees)

-- ===================== AIM LOCK =====================
local aimActive = false
local lockedTarget = nil

local function getClosestPlayer()
local closestPlayer = nil
local shortestDistance = math.huge
local currentCharacter = LocalPlayer.Character
if not currentCharacter then return nil end
local myRoot = currentCharacter:FindFirstChild("HumanoidRootPart")
if not myRoot then return nil end

for _, player in ipairs(Players:GetPlayers()) do  
    if player ~= LocalPlayer and player.Character then  
        local enemyRoot = player.Character:FindFirstChild("HumanoidRootPart")  
        local enemyHumanoid = player.Character:FindFirstChildOfClass("Humanoid")  
        if enemyRoot and enemyHumanoid and enemyHumanoid.Health > 0 then  
            local dist = (myRoot.Position - enemyRoot.Position).Magnitude  
            if dist < shortestDistance then  
                shortestDistance = dist  
                closestPlayer = player.Character  
            end  
        end  
    end  
end  
return closestPlayer

end

local root = char:FindFirstChild("HumanoidRootPart")
local hum = char:FindFirstChildOfClass("Humanoid")
if not root or not hum then return end

local myPos = root.Position  
            local totalRepulsion = Vector3.new(0, 0, 0)  
            local threatCount = 0  
            local maxThreatDist = 150  

            for _, player in ipairs(Players:GetPlayers()) do  
                if player ~= LocalPlayer and player.Character then  
                    local enemyRoot = player.Character:FindFirstChild("HumanoidRootPart")  
                    local enemyHum = player.Character:FindFirstChildOfClass("Humanoid")  
                    if enemyRoot and enemyHum and enemyHum.Health > 0 then  
                        local enemyPos = enemyRoot.Position  
                        local diff = myPos - enemyPos  
                        local dist = diff.Magnitude  
                          
                        if dist < maxThreatDist and dist > 0.5 then  
                            local weight = 1 / (dist * dist)  
                            local awayDir = Vector3.new(diff.X, 0, diff.Z).Unit  
                            totalRepulsion = totalRepulsion + (awayDir * weight * 1000)  
                            threatCount = threatCount + 1  
                        end  
                    end  
                end  
            end  

            if threatCount > 0 and totalRepulsion.Magnitude > 0 then  
                local finalDir = totalRepulsion.Unit  
                hum:Move(finalDir, true)  
                local targetCFrame = CFrame.lookAt(myPos, myPos + finalDir)  
                root.CFrame = root.CFrame:Lerp(targetCFrame, 0.3)  
                  
                local currentSpeed = math.max(hum.WalkSpeed, 40)  
                root.AssemblyLinearVelocity = Vector3.new(  
                    finalDir.X * currentSpeed,  
                    root.AssemblyLinearVelocity.Y,  
                    finalDir.Z * currentSpeed  
                )  
            else  
                hum:Move(Vector3.new(0, 0, 0), false)  
            end  
        end)  
    end  
end

local freecamActive = false
local freecamPos = Vector3.new(0, 0, 0)
local freecamSpeed = 1.5
local freecamJoystickGui = nil
local joystickVector = Vector2.new(0, 0)

local function createFreecamJoystick()
if freecamJoystickGui then
freecamJoystickGui.Visible = true
return
end

freecamJoystickGui = Instance.new("Frame")  
freecamJoystickGui.Name = "FreecamJoystickGui"  
freecamJoystickGui.Size = UDim2.new(0, 130, 0, 130)  
freecamJoystickGui.Position = UDim2.new(0, 8, 0.5, -65)  
freecamJoystickGui.BackgroundColor3 = Color3.fromRGB(8, 12, 10)  
freecamJoystickGui.BackgroundTransparency = 0.4  
freecamJoystickGui.Parent = ScreenGui  

local outerCorner = Instance.new("UICorner")  
outerCorner.CornerRadius = UDim.new(1, 0)  
outerCorner.Parent = freecamJoystickGui  

local stick = Instance.new("Frame")  
stick.Name = "Stick"  
stick.Size = UDim2.new(0, 50, 0, 50)  
stick.Position = UDim2.new(0.5, -25, 0.5, -25)  
stick.BackgroundColor3 = Color3.fromRGB(255, 255, 255)  
stick.BackgroundTransparency = 0.3  
stick.Parent = freecamJoystickGui  

local stickCorner = Instance.new("UICorner")  
stickCorner.CornerRadius = UDim.new(1, 0)  
stickCorner.Parent = stick  

local dragging = false  
local touchInput = nil  

local function updateJoystick(input)  
    local center = freecamJoystickGui.AbsolutePosition + freecamJoystickGui.AbsoluteSize / 2  
    local delta = Vector2.new(input.Position.X, input.Position.Y) - center  
    local maxRadius = freecamJoystickGui.AbsoluteSize.X / 2  
    local clampedDelta = delta.Magnitude > maxRadius and delta.Unit * maxRadius or delta  
      
    stick.Position = UDim2.new(0.5, clampedDelta.X - 25, 0.5, clampedDelta.Y - 25)  
    joystickVector = Vector2.new(clampedDelta.X / maxRadius, -clampedDelta.Y / maxRadius)  
end  

freecamJoystickGui.InputBegan:Connect(function(input)  
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then  
        dragging = true  
        touchInput = input  
        updateJoystick(input)  
    end  
end)  

UserInputService.InputChanged:Connect(function(input)  
    if dragging and (input == touchInput or input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then  
        updateJoystick(input)  
    end  
end)  

local function endDrag(input)  
    if input == touchInput or input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then  
        dragging = false  
        touchInput = nil  
        joystickVector = Vector2.new(0, 0)  
        TweenService:Create(stick, TweenInfo.new(0.15), {Position = UDim2.new(0.5, -25, 0.5, -25)}):Play()  
    end  
end  

UserInputService.InputEnded:Connect(endDrag)

end

        Camera.CameraSubject = char:FindFirstChildOfClass("Humanoid")

RunService.RenderStepped:Connect(function(dt)
if freecamActive and Camera then
local moveDir = Vector3.new()

if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + Camera.CFrame.LookVector end  
    if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - Camera.CFrame.LookVector end  
    if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - Camera.CFrame.RightVector end  
    if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + Camera.CFrame.RightVector end  
    if UserInputService:IsKeyDown(Enum.KeyCode.E) then moveDir = moveDir + Vector3.new(0, 1, 0) end  
    if UserInputService:IsKeyDown(Enum.KeyCode.Q) then moveDir = moveDir - Vector3.new(0, 1, 0) end  

    if joystickVector.Magnitude > 0 then  
        moveDir = moveDir + (Camera.CFrame.RightVector * joystickVector.X) + (Camera.CFrame.LookVector * joystickVector.Y)  
    end  

    if moveDir.Magnitude > 0 then  
        freecamPos = freecamPos + moveDir.Unit * (freecamSpeed * 50 * dt)  
    end  
    Camera.CFrame = CFrame.new(freecamPos, freecamPos + Camera.CFrame.LookVector)  
end

end)
